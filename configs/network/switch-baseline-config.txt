! NETWORK SWITCH BASELINE CONFIGURATION (CISCO IOS)
! ------------------------------------------------------------------------------

! 1. GENERAL SYSTEM SETTINGS
! ------------------------------------------------------------------------------
hostname Switch-01
! Prevent typo'd commands from attempting DNS lookups (saves time)
no ip domain-lookup
! Encrypt legacy passwords in config file
service password-encryption

! 2. ACCOUNT SECURITY
! ------------------------------------------------------------------------------
! Enforce minimum password length
security passwords min-length 10
! Create admin user (Use 'secret', never 'password')
username admin privilege 15 secret [YOUR_COMPLEX_PASSWORD]
! Encrypt the Enable password
enable secret [YOUR_ENABLE_PASSWORD]

! 3. SSH CONFIGURATION (Kill Telnet)
! ------------------------------------------------------------------------------
ip domain-name internal.corp
crypto key generate rsa modulus 2048
ip ssh version 2
ip ssh time-out 60
ip ssh authentication-retries 3

! Apply to VTY Lines (Remote Access)
line vty 0 15
 transport input ssh
 logging synchronous
 login local
 exec-timeout 5 0  ! Disconnect after 5 mins idle
exit

! 4. CONSOLE SECURITY
! ------------------------------------------------------------------------------
line console 0
 logging synchronous
 login local
 exec-timeout 5 0
exit

! 5. SECURITY & HARDENING
! ------------------------------------------------------------------------------
! Banner to discourage unauthorized access (Legal requirement)
banner motd ^C
*************************************************************************
* WARNING: UNATHORIZED ACCESS PROHIBITED.                              *
* All activity is monitored and logged.                                *
*************************************************************************
^C

! Disable unused services
no service pad
no ip http server
no ip http secure-server
no cdp run  ! Disable CDP globally to hide device info (Optional: enable per port)

! 6. TIME SYNCHRONIZATION (NTP)
! ------------------------------------------------------------------------------
! Critical for log correlation
ntp server 0.pool.ntp.org
service timestamps log datetime msec
service timestamps debug datetime msec

! 7. SPANNING TREE (Loop Prevention)
! ------------------------------------------------------------------------------
! Use Rapid-PVST for faster convergence
spanning-tree mode rapid-pvst
! Enable PortFast/BPDUGuard by default on edge ports (prevents loops from users)
spanning-tree portfast default
spanning-tree portfast bpduguard default
